#include <iostream>
#include <string>

bool DivisionByZeroCheck(int* res, int* arg1, int* arg2) {
   
try {
    if (0==*arg2) {
        throw runtime_error("cannot divide by 0");
    }
else { *res = ( *arg1/ *arg2); }
}
catch (const runtime_error& ErrorMessage) {
    cout << ErrorMessage.what();
    return false;
}
return true;

}
