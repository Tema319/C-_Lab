#include <iostream>
#include <string>
using namespace std;

int Task = 1;

class Fraction {
public:
    Fraction(int numerator, int denominator);
    string toString();
    double toDouble();
    Fraction plus(Fraction that);
    Fraction minus(Fraction that);
    Fraction times(Fraction that);
    Fraction by(Fraction that);

private:
    int numerator;
    int denominator;
    bool SetNumToNegativeAfterRecursion = false;
    bool SetDenToNegativeAfterRecursion = false;
    void reduce();
};

Fraction::Fraction(int numerator, int denominator) {
    this->numerator = numerator;
    this->denominator = denominator;
}

string Fraction::toString() {

    string FinalString;

    if (this->numerator == 0) { return "0"; }
    if (this->denominator == 0) { return to_string(this->numerator); }

    int Negatives = 0;
    int EntirelyPositiveNumerator = this->numerator;
    int EntirelyPositiveDenominator = this->denominator;

    if (this->numerator < 0) {
        Negatives++;
        EntirelyPositiveNumerator = (this->numerator * -1);
    }

    if (this->denominator < 0) {
        Negatives++;
        EntirelyPositiveDenominator = (this->denominator * -1);
    }

    if (EntirelyPositiveNumerator > EntirelyPositiveDenominator) {

        int RemainderOffset = EntirelyPositiveNumerator % EntirelyPositiveDenominator;
        int ApplicableDividend = (EntirelyPositiveNumerator - RemainderOffset);
        int AmountOfFills = ApplicableDividend / EntirelyPositiveDenominator;

        string StringFills = to_string(AmountOfFills);
        string StringRemainderOffset = to_string(RemainderOffset);
        string StringPositiveDenominator = to_string(EntirelyPositiveDenominator);

        string FractionPortion = "";

        if (RemainderOffset != 0) {
            FinalString = StringFills + " " + StringRemainderOffset + "/" + StringPositiveDenominator;
        }
        else {
            FinalString = StringFills;
        }
    }
    else {
        string StringNumerator = to_string(EntirelyPositiveNumerator);
        string StringDenominator = to_string(EntirelyPositiveDenominator);
        FinalString = StringNumerator + "/" + StringDenominator;
    }

    if (Negatives % 2 != 0) {
        FinalString = "-" + FinalString;
    }

    return FinalString;
}

double Fraction::toDouble() {
    double NumeratorAdjusted = (double)this->numerator;
    double DenominatorAdjusted = (double)this->denominator;
    double InDoubleFormatNumber = NumeratorAdjusted / DenominatorAdjusted;
    return InDoubleFormatNumber;
}

Fraction Fraction::plus(Fraction that) {
    int num1 = this->numerator * that.denominator;
    int num2 = that.numerator * this->denominator;
    int num = num1 + num2;
    int den = this->denominator * that.denominator;

    Fraction result(num, den);
    result.reduce();

    return result;
}

Fraction Fraction::minus(Fraction that) {
    int num1 = this->numerator * that.denominator;
    int num2 = that.numerator * this->denominator;
    int num = num1 - num2;
    int den = this->denominator * that.denominator;

    Fraction result(num, den);
    result.reduce();

    return result;
}

Fraction Fraction::times(Fraction that) {
    int SavedNumerator = this->numerator;
    int SavedDenominator = this->denominator;

    int num = SavedNumerator * that.numerator;
    int den = SavedDenominator * that.denominator;

    Fraction result(num, den);
    result.reduce();

    return result;
}

Fraction Fraction::by(Fraction that) {
    int num = this->numerator * that.denominator;
    int den = this->denominator * that.numerator;

    Fraction result(num, den);
    result.reduce();

    return result;
}

void Fraction::reduce() {

    if (this->numerator < 0) {
        this->SetNumToNegativeAfterRecursion = true;
        this->numerator = (this->numerator * -1);
    }

    if (this->denominator < 0) {
        this->SetDenToNegativeAfterRecursion = true;
        this->denominator = (this->denominator * -1);
    }

    int LowerNumber;

    if (this->numerator < this->denominator) {
        LowerNumber = this->numerator;
    }
    else {
        LowerNumber = this->denominator;
    }

    bool ContinueRecursion = false;

    if (LowerNumber >= 2) {
        for (int i = 2; i <= LowerNumber; i++) {

            if (this->numerator <= 1 || this->denominator <= 1) {
                ContinueRecursion = false;
                break;
            }
            else if ((this->numerator % i) == 0 && (this->denominator % i) == 0) {
                this->numerator = (this->numerator / i);
                this->denominator = (this->denominator / i);
                ContinueRecursion = true;
                break;
            }
        }
    }
    else {
        ContinueRecursion = false;
    }

    if (ContinueRecursion == true) {
        this->reduce();
    }
    else {
        if (this->SetNumToNegativeAfterRecursion == true) {
            this->numerator = (this->numerator * -1);
        }

        if (this->SetDenToNegativeAfterRecursion == true) {
            this->denominator = (this->denominator * -1);
        }
    }
}

        }

            StringNum1 = StringNum1 + input1[i];
        }
    }

    for (int i = Offset1; i < input1.size(); i++) {

        if (input1[i] == '-') {
            StringDen1 = "-" + StringDen1;
        }
        else if (isdigit(input1[i])) {
            StringDen1 = StringDen1 + input1[i];
        }
    }

    IntNum1 = stoi(StringNum1);
    IntDen1 = stoi(StringDen1);

    int Offset2 = 0;

    for (int i = 0; i < input2.size(); i++) {

        if (input2[i] == '/') {
            Offset2 = i + 1;
            break;
        }

        if (input2[i] == '-') {
            StringNum2 = "-" + StringNum2;
        }
        else if (isdigit(input2[i])) {

        }
    }

    for (int i = Offset2; i < input2.size(); i++) {

        if (input2[i] == '-') {
            StringDen2 = "-" + StringDen2;
        }
        else if (isdigit(input2[i])) {
            StringDen2 = StringDen2 + input2[i];
        }
    }

    IntNum2 = stoi(StringNum2);

    Fraction FirstFraction = Fraction(IntNum1, IntDen1);
